pragma solidity 0.8.28;

import {IVerifier} from "../chain-interfaces/IVerifier.sol";
import {IZiskVerifier} from "../chain-interfaces/IZiskVerifier.sol";
import {ISnarkPlonkVerifier} from "../chain-interfaces/ISnarkPlonkVerifier.sol";

/// @title ZiSK Verifier
/// @author Matter Labs
/// @custom:security-contact security@matterlabs.dev
/// @notice Verifies ZiSK SNARK proofs as an inner verifier for MultiProofVerifier.
/// @dev Implements IZiskVerifier so it can be registered via setZiskVerifier()
///      and its pinned wire-form VKs can be read back for range binding.
///      The MultiProofVerifier strips the proof header (type+version, previous_hash, N,
///      airbender_proof) before calling this contract, so _proof starts at ZiSK data:
///        _proof[0..24]  = ZiSK SNARK proof (24 uint256 = 768 bytes)
///        _proof[24..34] = ZiSK public values (10 uint256 = 320 bytes):
///          programVK(32) || guest publics(256) || rootCVadcopFinal(32)
///
/// @dev Generated by `cargo run -- --variant zisk` in `tools/`.
///      VK hash: 0x{{vk_hash}}
contract ZiskVerifier is IZiskVerifier {
    /// @notice The snarkJS-generated Plonk verifier for the ZiSK vadcop-final
    ///         wrap. Regenerated with the circuit, so it is deployed as a
    ///         standalone contract and referenced by address; see
    ///         `verifiers/README.md` for the generation and deployment flow.
    ISnarkPlonkVerifier public immutable PLONK_VERIFIER;

    constructor(ISnarkPlonkVerifier _plonkVerifier) {
        PLONK_VERIFIER = _plonkVerifier;
    }

    /// @notice ZiSK ELF-specific verification key (programVK).
    /// Derived from the ROM Merkle root of the guest ELF binary.
{{program_vk_constants}}
    /// @notice ZiSK vadcop final root commitment.
{{root_cv_constants}}
    /// @notice BN254 scalar field modulus.
    uint256 private constant _RFIELD =
        21888242871839275222246405745257275088548364400416034343698204186575808495617;

    /// @notice Verification key hash (must match server's ProvingVersion VK hash).
    bytes32 private constant _VK_HASH =
        0x{{vk_hash}};

    /// @inheritdoc IZiskVerifier
    function programVK() public pure override returns (bytes32) {
        return
            bytes32(
                (uint256(_PROGRAM_VK_0) << 192) |
                    (uint256(_PROGRAM_VK_1) << 128) |
                    (uint256(_PROGRAM_VK_2) << 64) |
                    uint256(_PROGRAM_VK_3)
            );
    }

    /// @inheritdoc IZiskVerifier
    function rootCVadcopFinal() public pure override returns (bytes32) {
        return
            bytes32(
                (uint256(_ROOT_CV_ADCOP_FINAL_0) << 192) |
                    (uint256(_ROOT_CV_ADCOP_FINAL_1) << 128) |
                    (uint256(_ROOT_CV_ADCOP_FINAL_2) << 64) |
                    uint256(_ROOT_CV_ADCOP_FINAL_3)
            );
    }

    /// @inheritdoc IVerifier
    function verify(
        uint256[] calldata _publicInputs,
        uint256[] calldata _proof
    ) external view override returns (bool) {
        require(_proof.length >= 34, "ZiskVerifier: proof too short");

        // Extract 24-element ZiSK SNARK proof.
        uint256[24] memory ziskProof;
        for (uint256 i = 0; i < 24; i++) {
            ziskProof[i] = _proof[i];
        }

        // The public values must open with this contract's programVK and
        // close with its rootCVadcopFinal — otherwise the proof attests to a
        // different guest program or SNARK circuit.
        if (_proof[24] != uint256(programVK()) || _proof[33] != uint256(rootCVadcopFinal())) {
            return false;
        }

        // Extract the 320-byte ZiSK public values
        // (programVK || guest publics || rootCVadcopFinal) from 10 uint256 elements.
        bytes memory publicValues = new bytes(320);
        for (uint256 i = 0; i < 10; i++) {
            uint256 val = _proof[24 + i];
            assembly {
                mstore(add(add(publicValues, 32), mul(i, 32)), val)
            }
        }

        // The circuit's single public signal: sha256(publicValues) % RFIELD.
        uint256 publicValuesDigest = uint256(sha256(publicValues)) % _RFIELD;

        return PLONK_VERIFIER.verifyProof(ziskProof, [publicValuesDigest]);
    }

    /// @inheritdoc IVerifier
    function verificationKeyHash() external pure override returns (bytes32) {
        return _VK_HASH;
    }
}
